# Rollup and Rolldown

## Rollup

```ts [rollup.config.ts]
import compdi from "compdi/plugin/rollup";

export default {
  input: "src/index.ts",
  plugins: [compdi()],
};
```

## Rolldown

```ts [rolldown.config.ts]
import compdi from "compdi/plugin/rolldown";

export default {
  input: "src/index.ts",
  plugins: [compdi()],
};
```
